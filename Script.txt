-- ============================================================
-- KEY SYSTEM (HASHED — raw keys never appear in source)
-- ============================================================
-- Generate hashes with SHA256(salt .. key), where salt = KEY_SALT.
-- Then paste into HASH_DATABASE keyed by the ENTERED KEY STRING.
-- We look up by key-string, then verify the assigned UserId matches.
--
-- Format: HASH_DATABASE[key_string] = { userId = <number>, hash = "<hex>" }
--
-- You may instead store only the hash and derive the UserId from a
-- separate table; both are shown. Keeping userId in the entry makes
-- the "assigned UserId" check trivial.
-- ============================================================

local KEY_SALT = "s0urce"

-- Minimal pure-Lua SHA-256 (bit32). Public-domain implementation.
local function sha256(msg)
    local band, bxor, rrot = bit32.band, bit32.bxor, bit32.rrotate
    local h0,h1,h2,h3,h4,h5,h6,h7 =
        0x6a09e667,0xbb67ae85,0x3c6ef372,0xa54ff53a,
        0x510e527f,0x9b05688c,0x1f83d9ab,0x5be0cd19
    local K = {
        0x428a2f98,0x71374491,0xb5c0fbcf,0xe9b5dba5,0x3956c25b,0x59f111f1,0x923f82a4,0xab1c5ed5,
        0xd807aa98,0x12835b01,0x243185be,0x550c7dc3,0x72be5d74,0x80deb1fe,0x9bdc06a7,0xc19bf174,
        0xe49b69c1,0xefbe4786,0x0fc19dc6,0x240ca1cc,0x2de92c6f,0x4a7484aa,0x5cb0a9dc,0x76f988da,
        0x983e5152,0xa831c66d,0xb00327c8,0xbf597fc7,0xc6e00bf3,0xd5a79147,0x06ca6351,0x14292967,
        0x27b70a85,0x2e1b2138,0x4d2c6dfc,0x53380d13,0x650a7354,0x766a0abb,0x81c2c92e,0x92722c85,
        0xa2bfe8a1,0xa81a664b,0xc24b8b70,0xc76c51a3,0xd192e819,0xd6990624,0xf40e3585,0x106aa070,
        0x19a4c116,0x1e376c08,0x2748774c,0x34b0bcb5,0x391c0cb3,0x4ed8aa4a,0x5b9cca4f,0x682e6ff3,
        0x748f82ee,0x78a5636f,0x84c87814,0x8cc70208,0x90befffa,0xa4506ceb,0xbef9a3f7,0xc67178f2,
    }
    local function tobytes(s)
        local t = {}
        for i = 1, #s do t[i] = s:byte(i) end
        return t
    end
    local bytes = tobytes(msg)
    local bitlen = #bytes * 8
    bytes[#bytes+1] = 0x80
    while (#bytes % 64) ~= 56 do bytes[#bytes+1] = 0 end
    for i = 7, 0, -1 do
        bytes[#bytes+1] = math.floor(bitlen / (2^(8*i))) % 256
    end
    local function word(a,b,c,d) return a*0x1000000 + b*0x10000 + c*0x100 + d end
    local h = {h0,h1,h2,h3,h4,h5,h6,h7}
    for chunk = 0, (#bytes/64) - 1 do
        local w = {}
        local base = chunk*64
        for i = 0, 15 do
            w[i] = word(bytes[base+i*4+1], bytes[base+i*4+2], bytes[base+i*4+3], bytes[base+i*4+4])
        end
        for i = 16, 63 do
            local s0 = bxor(rrot(w[i-15],7), rrot(w[i-15],18), math.floor(w[i-15]/0x200000))
            local s1 = bxor(rrot(w[i-2],17), rrot(w[i-2],19), math.floor(w[i-2]/0x4000))
            w[i] = (w[i-16] + s0 + w[i-7] + s1) % 0x100000000
        end
        local a,b,c,d,e,f,g,hh = h[1],h[2],h[3],h[4],h[5],h[6],h[7],h[8]
        for i = 0, 63 do
            local S1 = bxor(rrot(e,6), rrot(e,11), rrot(e,25))
            local ch = bxor(band(e,f), band(bxor(e,0xffffffff), g))
            local t1 = (hh + S1 + ch + K[i+1] + w[i]) % 0x100000000
            local S0 = bxor(rrot(a,2), rrot(a,13), rrot(a,22))
            local mj = bxor(band(a,b), band(a,c), band(b,c))
            local t2 = (S0 + mj) % 0x100000000
            hh=g; g=f; f=e; e=(d+t1)%0x100000000
            d=c; c=b; b=a; a=(t1+t2)%0x100000000
        end
        h[1]=(h[1]+a)%0x100000000; h[2]=(h[2]+b)%0x100000000
        h[3]=(h[3]+c)%0x100000000; h[4]=(h[4]+d)%0x100000000
        h[5]=(h[5]+e)%0x100000000; h[6]=(h[6]+f)%0x100000000
        h[7]=(h[7]+g)%0x100000000; h[8]=(h[8]+hh)%0x100000000
    end
    local out = {}
    for i = 1, 8 do out[i] = string.format("%08x", h[i]) end
    return table.concat(out)
end

-- ============================================================
-- HASH DATABASE
-- ============================================================
-- Each entry is keyed by the SHA-256 hash of (KEY_SALT .. raw_key).
-- The value is the UserId that this key is assigned to.
--
-- HOW TO FILL:
--   1. For each key, compute SHA256("s0urce" .. key)
--      (use https://emn178.github.io/online-tools/sha256.html)
--   2. Paste the 64-char hex hash as the table key.
--   3. Set the value to the assigned Roblox UserId (number).
--
-- The raw keys (KEYSYSTEM001, etc.) DO NOT appear anywhere below.
-- ============================================================

local HASH_DATABASE = {
    -- ["<sha256-hex>"] = <UserId>,
    --
    -- Paste your generated hashes here. Example shape:
    -- ["aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa"] = 11661175541,
    -- ["bbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbb"] = 11569705587,
    --
    -- PLACEHOLDERS (DELETE and replace with real hashes):
    [sha256(KEY_SALT .. "KEYSYSTEM001")]   = 11661175541,
    [sha256(KEY_SALT .. "KEYSYSTEM99910")] = 11569705587,
    [sha256(KEY_SALT .. "KEYSYSTEM91935")] = 1000001,
    [sha256(KEY_SALT .. "KEYSYSTEM99750")] = 1000002,
    [sha256(KEY_SALT .. "KEYSYSTEM11838")] = 1000003,
    [sha256(KEY_SALT .. "KEYSYSTEM99024")] = 1000004,
    [sha256(KEY_SALT .. "KEYSYSTEM0018")]  = 1000005,
    [sha256(KEY_SALT .. "KEYSYSTEM72748")] = 1000006,

-- NOTE ON THE ABOVE:
-- The lines like [sha256(KEY_SALT .. "KEYSYSTEM001")] still reveal the
-- raw key at RUNTIME (when the file is parsed). This defeats the point.
--
-- After you generate your hashes, REPLACE those lines with literal
-- hex strings:
--     ["<64-hex-char-hash>"] = 11661175541,
--     ["<64-hex-char-hash>"] = 11569705587,
--     ...
-- Do NOT leave the sha256(...) calls in — they leak the keys.
--
-- (They are shown here only so the script still runs out-of-the-box
-- and you can test the flow before generating real hashes.)

local INFINITE_FOR_ASSIGNED_USER = true
local USED_KEYS_FILE = "galaxy_sniper_used_keys.json"

local function readUsedKeys()
    local used = {}
    if typeof(readfile) == "function" and typeof(isfile) == "function" then
        local ok1, exists = pcall(isfile, USED_KEYS_FILE)
        if ok1 and exists then
            local ok2, contents = pcall(readfile, USED_KEYS_FILE)
            if ok2 and contents and contents ~= "" then
                local ok3, decoded = pcall(function()
                    return HttpService:JSONDecode(contents)
                end)
                if ok3 and type(decoded) == "table" then
                    used = decoded
                end
            end
        end
    end
    if _G.__GalaxySniperUsedKeys then
        for k, v in pairs(_G.__GalaxySniperUsedKeys) do
            used[k] = v
        end
    end
    return used
end

local function saveUsedKeys(used)
    _G.__GalaxySniperUsedKeys = used
    if typeof(writefile) == "function" then
        local ok, encoded = pcall(function()
            return HttpService:JSONEncode(used)
        end)
        if ok then
            pcall(writefile, USED_KEYS_FILE, encoded)
        end
    end
end

local function isOurGui(instance)
    local p = instance
    for _ = 1, 10 do
        if not p then break end
        if p.Name == "SourcesHubRedeemerGui" then return true end
        p = p.Parent
    end
    return false
end

local function findAllTextBoxes(pg)
    local boxes = {}
    for _, gui in ipairs(pg:GetChildren()) do
        if gui:IsA("ScreenGui") and gui.Enabled and gui.Name ~= "SourcesHubRedeemerGui" then
            for _, d in ipairs(gui:GetDescendants()) do
                if d:IsA("TextBox") and not isOurGui(d) then
                    boxes[#boxes+1] = d
                end
            end
        end
    end
    return boxes
end

local function isVisibleChain(inst)
    local current = inst
    while current do
        if current:IsA("GuiObject") then
            if not current.Visible then return false end
        end
        if current:IsA("ScreenGui") then
            if not current.Enabled then return false end
            return true
        end
        current = current.Parent
    end
    return true
end

local function findCodeButtons(pg)
    local btns = {}
    for _, gui in ipairs(pg:GetChildren()) do
        if gui:IsA("ScreenGui") and gui.Enabled and gui.Name ~= "SourcesHubRedeemerGui" then
            for _, d in ipairs(gui:GetDescendants()) do
                if (d:IsA("TextButton") or d:IsA("ImageButton")) and not isOurGui(d) then
                    local n  = d.Name:lower()
                    local pn = (d.Parent and d.Parent.Name or ""):lower()
                    if (n:find("code") or n:find("redeem") or pn:find("code") or pn:find("redeem"))
                        and isVisibleChain(d) then
                        btns[#btns+1] = d
                    end
                end
            end
        end
    end
    return btns
end

local function clickButton(btn)
    if not btn then return false end
    local methods = {}

    methods[#methods+1] = function() btn.MouseButton1Click:Fire() end
    methods[#methods+1] = function() btn.Activated:Fire() end

    if typeof(firesignal) == "function" then
        methods[#methods+1] = function() firesignal(btn.MouseButton1Click) end
        methods[#methods+1] = function() firesignal(btn.Activated) end
    end

    if typeof(getconns) == "function" then
        methods[#methods+1] = function()
            local ok, cs = pcall(getconns, btn.MouseButton1Click)
            if ok and type(cs) == "table" then
                for _, c in ipairs(cs) do pcall(function() c:Fire() end) end
            end
            local ok2, cs2 = pcall(getconns, btn.Activated)
            if ok2 and type(cs2) == "table" then
                for _, c in ipairs(cs2) do pcall(function() c:Fire() end) end
            end
        end
    end

    if typeof(fireclick) == "function" then
        methods[#methods+1] = function() fireclick(btn) end
    end

    local anyOk = false
    for _, fn in ipairs(methods) do
        local ok = pcall(fn)
        anyOk = anyOk or ok
    end
    return anyOk
end

local function writeCodeToBox(box, code)
    if not box then return false end
    pcall(function()
        box.Text = code
    end)
    return box.Text == code
end

local function fireBoxFocusLost(box)
    if not box then return false end
    local anyFired = false

    if typeof(firesignal) == "function" then
        local ok = pcall(firesignal, box.FocusLost, true)
        anyFired = anyFired or ok
    end

    if typeof(getconns) == "function" then
        local ok, cs = pcall(getconns, box.FocusLost)
        if ok and type(cs) == "table" then
            for _, c in ipairs(cs) do
                local fn
                pcall(function() fn = c.Function end)
                if fn and typeof(getupvalues) == "function" and typeof(setupv) == "function" then
                    local uOk, ups = pcall(getupvalues, fn)
                    if uOk and type(ups) == "table" then
                        for i, v in pairs(ups) do
                            if type(v) == "boolean" and v == true then
                                pcall(setupv, fn, i, false)
                            end
                        end
                    end
                end
                local fOk = pcall(function()
                    if c.Enabled ~= false then c:Fire(true) end
                end)
                anyFired = anyFired or fOk
            end
        end
    end

    return anyFired
end

local function typeAndSubmitCode(code)
    if not LP then return false, "no LP" end
    local pg = LP:FindFirstChildOfClass("PlayerGui")
    if not pg then return false, "no PlayerGui" end

    local codesGui = pg:FindFirstChild("Codes")
    if codesGui then
        if codesGui:IsA("ScreenGui") then
            codesGui.Enabled = true
        end
        local codesFrame = codesGui:FindFirstChild("Codes") or codesGui
        if codesFrame then
            if codesFrame:IsA("GuiObject") then
                codesFrame.Visible = true
            end
            local cur = codesFrame
            while cur and cur ~= codesGui do
                if cur:IsA("GuiObject") then cur.Visible = true end
                cur = cur.Parent
            end

            local box = nil
            for _, d in ipairs(codesFrame:GetDescendants()) do
                if d:IsA("TextBox") and not isOurGui(d) then
                    box = d
                    break
                end
            end

            local submitBtn = nil
            for _, d in ipairs(codesFrame:GetDescendants()) do
                if (d:IsA("TextButton") or d:IsA("ImageButton")) and not isOurGui(d) then
                    local n = d.Name:lower()
                    local txt = ""
                    pcall(function() txt = d.Text:lower() end)
                    if n:find("submit") or txt:find("submit") or n:find("redeem") or txt:find("redeem") or n:find("claim") or txt:find("confirm") or n:find("enter") then
                        submitBtn = d
                        break
                    end
                end
            end
            if not submitBtn then
                for _, d in ipairs(codesFrame:GetDescendants()) do
                    if (d:IsA("TextButton") or d:IsA("ImageButton")) and not isOurGui(d) then
                        local n = d.Name:lower()
                        if not n:find("close") and not n:find("x") and not n:find("toggle") then
                            submitBtn = d
                            break
                        end
                    end
                end
            end

            if box then
                writeCodeToBox(box, code)
                task.wait(0.05)
                if submitBtn then
                    clickButton(submitBtn)
                end
                fireBoxFocusLost(box)
                return true, "submitted via PlayerGui.Codes.Codes"
            end
        end
    end

    local function tryOpenPanel()
        local btns = findCodeButtons(pg)
        for _, btn in ipairs(btns) do
            clickButton(btn)
            task.wait(0.05)
        end
        return #btns > 0
    end

    tryOpenPanel()
    task.wait(0.3)

    local box = nil
    local deadline = tick() + 3
    while tick() < deadline do
        local allBoxes = findAllTextBoxes(pg)
        for _, d in ipairs(allBoxes) do
            if isVisibleChain(d) then
                local n   = d.Name:lower()
                local pn  = (d.Parent and d.Parent.Name or ""):lower()
                if n:find("code") or pn:find("code") or n:find("redeem") or pn:find("redeem") or n:find("input") or pn:find("input") or n:find("textbox") or n:find("enter") then
                    box = d
                    break
                end
            end
        end
        if not box then
            for _, d in ipairs(allBoxes) do
                if isVisibleChain(d) then box = d; break end
            end
        end
        if box then break end
        task.wait(0.1)
    end

    if not box then return false, "no codebox visible" end

    writeCodeToBox(box, code)
    task.wait(0.05)

    local redeemBtn = nil
    local searchNames = {"submit","redeem","claim","confirm","enter","send","apply","ok","use","go","check"}
    local p = box.Parent
    for _ = 1, 8 do
        if not p then break end
        for _, d in ipairs(p:GetDescendants()) do
            if (d:IsA("TextButton") or d:IsA("ImageButton")) and not isOurGui(d) and d ~= box then
                local n = d.Name:lower()
                local txt = ""
                pcall(function() txt = d.Text:lower() end)
                for _, sn in ipairs(searchNames) do
                    if n:find(sn) or txt:find(sn) then
                        if isVisibleChain(d) then
                            redeemBtn = d
                            break
                        end
                    end
                end
                if redeemBtn then break end
            end
        end
        if redeemBtn then break end
        p = p.Parent
    end

    if redeemBtn then
        clickButton(redeemBtn)
    end

    fireBoxFocusLost(box)

    return true, "fallback methods used"

local function resolveNotifyRemote()
    if _G.PhiNotifyRemote then return _G.PhiNotifyRemote end
    local Net = ReplicatedStorage:WaitForChild("Packages"):WaitForChild("Net")
    local getinfo = debug and (debug.getinfo or debug.info)
    if getgc and getinfo and getconnections then
        for _, d in ipairs(Net:GetDescendants()) do
            if d:IsA("RemoteEvent") then
                local ok, cs = pcall(getconnections, d.OnClientEvent)
                if ok then
                    for _, c in ipairs(cs) do
                        local f, fn = pcall(function() return c.Function end)
                        if f and type(fn) == "function" then
                            local i, info = pcall(getinfo, fn)
                            if i and tostring(info.short_src or info.source or ""):find("NotificationController", 1, true) then
                                return d
                            end
                        end
                    end
                end
            end
        end
    end
    return nil
end

local function containsSammyKeyword(text)
    if not text or text == "" then return false end
    local lower = text:lower()
    for _, keyword in ipairs(sammyAnnouncementKeywords) do
        if lower:find(keyword, 1, true) then
            return true
        end
    end
    return false
end

-- Extract numbers from text, skipping any inside a "used" span.
local function extractNumberParts(text, usedSpans)
    if not text or text == "" then return "" end

    usedSpans = usedSpans or {}

    local function isUsed(pos)
        for _, span in ipairs(usedSpans) do
            if pos >= span.start and pos <= span["end"] then
                return true
            end
        end
        return false
    end

    local parts = {}

    local init = 1
    while true do
        local sepStart, sepEnd, sepWord, num = text:find("([%+%-]?%s*[%a&]*%s*)(%d+)", init)
        if not sepStart then break end

        local numStart = sepEnd - #num + 1

        if not isUsed(numStart) then
            local s = sepWord:gsub("%s+", ""):lower()
            if s == "-" then
                table.insert(parts, "-" .. num)
            else
                table.insert(parts, num)
            end
        end

        init = sepEnd + 1
    end

    return table.concat(parts)
end

-- Resolve a single riddle string (no "and" splitting) against sammyDatabase.
local function resolveSingleRiddle(text)
    if not text or text == "" then return nil end

    local lower = text:lower():gsub("^%s+", ""):gsub("%s+$", "")
    print("[DBG] input: [" .. lower .. "] len=" .. #lower)

    local candidates = {}

    -- ==== Literal scan: two-tier ====
    local bestContains = nil
    local bestInside   = nil

    for pattern, answer in pairs(sammyDatabase) do
        local containsPos = lower:find(pattern, 1, true)
        if containsPos then
            if not bestContains or #pattern > #bestContains.pattern then
                bestContains = {
                    pattern  = pattern,
                    answer   = answer,
                    position = containsPos,
                    length   = #pattern,
                }
            end
        else
            local insidePos = pattern:find(lower, 1, true)
            if insidePos then
                local ratio = #lower / #pattern
                if ratio >= 0.6 then
                    local extra = #pattern - #lower
                    if not bestInside
                       or extra < bestInside.extra
                       or (extra == bestInside.extra and #pattern > #bestInside.pattern) then
                        bestInside = {
                            pattern  = pattern,
                            answer   = answer,
                            position = 1,
                            length   = #lower,
                            extra    = extra,
                        }
                    end
                end
            end
        end
    end

    print("[DBG] bestContains=" .. tostring(bestContains and bestContains.pattern or "nil")
        .. " bestInside=" .. tostring(bestInside and (bestInside.pattern .. " extra=" .. bestInside.extra) or "nil"))

    if bestContains then
        table.insert(candidates, {
            pattern    = bestContains.pattern,
            answer     = bestContains.answer,
            position   = bestContains.position,
            length     = bestContains.length,
            isCombined = false,
            score      = 100000 + #bestContains.pattern,
        })
    elseif bestInside then
        table.insert(candidates, {
            pattern    = bestInside.pattern,
            answer     = bestInside.answer,
            position   = bestInside.position,
            length     = bestInside.length,
            isCombined = false,
            score      = 10000 + #bestInside.length - bestInside.extra,
        })
    end

    -- ==== Dynamic riddles ====
    local catPos = lower:find("cats name") or lower:find("cat name")
                   or lower:find("kitty name") or lower:find("pet name")
                   or lower:find("kitten name")
    if catPos and (lower:find("3 times") or lower:find("x3")
                   or lower:find("3x") or lower:find("three times")) then
        table.insert(candidates, {
            pattern    = "cats name 3 times",
            answer     = "NovaNovaNova",
            position   = catPos,
            length     = #"cats name 3 times",
            isCombined = false,
            score      = 100000 + #"cats name 3 times",
        })
    end

    local colorPos = lower:find("favorite color") or lower:find("fav color")
                     or lower:find("favourite color") or lower:find("colour")
    if colorPos and (lower:find("3 times") or lower:find("x3")
                     or lower:find("3x") or lower:find("three times")) then
        table.insert(candidates, {
            pattern    = "favorite color 3 times",
            answer     = "blueblueblue",
            position   = colorPos,
            length     = #"favorite color 3 times",
            isCombined = false,
            score      = 100000 + #"favorite color 3 times",
        })
    end

    -- ==== Combined mutation + enemy + favorite-color riddle ====
    local hasMutation = lower:find("mutation") ~= nil
    local hasEnemy    = lower:find("enemy") ~= nil
    local hasColor    = (lower:find("favorite color") or lower:find("fav color")
                         or lower:find("favorite colour") or lower:find("colour")) ~= nil

    if hasMutation and hasEnemy and hasColor then
        local hasSpecificMatch = false
        for _, c in ipairs(candidates) do
            if #c.pattern >= 18
               and (c.pattern:find("mutation") or c.pattern:find("enemy")
                    or c.pattern:find("favorite color") or c.pattern:find("color")) then
                hasSpecificMatch = true
                break
            end
        end

        if not hasSpecificMatch then
            local posMutation = lower:find("mutation")
            local posEnemy    = lower:find("enemy")
            local posColor    = lower:find("favorite color") or lower:find("fav color")
                                or lower:find("favorite colour") or lower:find("colour")

            local parts = {
                { pos = posMutation, val = "candy"  },
                { pos = posEnemy,    val = "jandel" },
                { pos = posColor,    val = "blue"   },
            }
            table.sort(parts, function(a, b)
                return (a.pos or 9999) < (b.pos or 9999)
            end)

            local combinedAns = parts[1].val .. parts[2].val .. parts[3].val
            local combinedPos = math.min(posMutation or 9999, posEnemy or 9999, posColor or 9999)

            table.insert(candidates, {
                pattern    = "combined mutation enemy color",
                answer     = combinedAns,
                position   = combinedPos,
                length     = 0,
                isCombined = true,
                score      = 0,
            })
        end
    end

    -- ==== Word-overlap fallback ====
    if #candidates == 0 then
        local stopWords = {
            ["the"]=true, ["and"]=true, ["was"]=true, ["what"]=true, ["whats"]=true,
            ["what's"]=true, ["when"]=true, ["did"]=true, ["does"]=true, ["is"]=true,
            ["are"]=true, ["my"]=true, ["of"]=true, ["in"]=true, ["on"]=true,
            ["to"]=true, ["a"]=true, ["an"]=true, ["for"]=true, ["with"]=true,
            ["it"]=true, ["he"]=true, ["she"]=true, ["we"]=true, ["they"]=true,
            ["i"]=true, ["you"]=true, ["be"]=true, ["been"]=true, ["has"]=true,
            ["had"]=true, ["do"]=true, ["that"]=true, ["this"]=true,
        }
        local function wordSet(str)
            local set = {}
            for w in str:gmatch("%a+") do
                if #w >= 3 and not stopWords[w] then
                    set[w] = true
                end
            end
            return set
        end
        local function wordOverlapScore(a, b)
            local inter, uni = 0, 0
            for w in pairs(a) do
                uni = uni + 1
                if b[w] then inter = inter + 1 end
            end
            for w in pairs(b) do
                if not a[w] then uni = uni + 1 end
            end
            if uni == 0 then return 0 end
            return inter / uni
        end

        local announceWords = wordSet(lower)
        for pattern, answer in pairs(sammyDatabase) do
            local patternWords = wordSet(pattern)
            local score = wordOverlapScore(announceWords, patternWords)
            if score >= 0.6 then
                table.insert(candidates, {
                    pattern    = pattern,
                    answer     = answer,
                    position   = 1,
                    length     = #pattern,
                    isCombined = false,
                    score      = score * #pattern,
                })
            end
        end
    end

    if #candidates == 0 then
        local numberPart = extractNumberParts(text, {})
        if numberPart ~= "" then
            return numberPart
        end
        return nil
    end

    table.sort(candidates, function(a, b)
        if a.position ~= b.position then
            return a.position < b.position
        end
        if (a.score or 0) ~= (b.score or 0) then
            return (a.score or 0) > (b.score or 0)
        end
        if a.isCombined ~= b.isCombined then
            return not a.isCombined
        end
        return a.pattern < b.pattern
    end)

    local accepted = {}
    local lastEnd = -1
    for _, c in ipairs(candidates) do
        if c.position > lastEnd then
            table.insert(accepted, c)
            lastEnd = c.position + c.length - 1
        end
    end

    local usedSpans = {}
    for _, c in ipairs(accepted) do
        table.insert(usedSpans, {
            start = c.position,
            ["end"] = c.position + c.length - 1,
        })
    end

    local numberPart = extractNumberParts(text, usedSpans)

    local result = ""
    for _, c in ipairs(accepted) do
        result = result .. c.answer
    end

    if numberPart ~= "" then
        result = result .. numberPart
    end

    print("[DBG] resolve result: [" .. tostring(result) .. "]")

    if result == "" then return nil end
    return result
end

-- Splits the announcement on " and " OR commas, resolves each segment, and
-- concatenates the answers.
local function searchDatabase(text)
    if not text or text == "" then return nil end

    local lower = text:lower()

    local segments = {}
    local start = 1
    while true do
        local andPos   = lower:find(" and ", start, true)
        local commaPos = lower:find(",", start, true)

        local cutPos, cutLen
        if andPos and (not commaPos or andPos <= commaPos) then
            cutPos, cutLen = andPos, 5
        elseif commaPos then
            cutPos, cutLen = commaPos, 1
        end

        if not cutPos then
            table.insert(segments, text:sub(start))
            break
        end

        table.insert(segments, text:sub(start, cutPos - 1))
        start = cutPos + cutLen
    end

    print("[DBG] searchDatabase raw: [" .. text .. "]")
    for i, s in ipairs(segments) do
        print("[DBG] segment " .. i .. ": [" .. s .. "]")
    end

    if #segments <= 1 then
        return resolveSingleRiddle(text)
    end

    local combined = ""
    local anyResolved = false
    for _, seg in ipairs(segments) do
        local trimmed = seg:match("^%s*(.-)%s*$")
        if trimmed ~= "" then
            local ans = resolveSingleRiddle(trimmed)
            if ans then
                combined = combined .. ans
                anyResolved = true
            end
        end
    end

    if anyResolved and combined ~= "" then
        return combined
    end

    return resolveSingleRiddle(text)
end

local function isValidCode(msg)
    return msg:match("^[A-Za-z0-9]+$") ~= nil
end

local function updatePreview()
    if #buffer == 0 then
        previewBox.Text = ""
        return
    end
    local count = math.min(#buffer, mode)
    local combined = table.concat(buffer, "", 1, count)
    previewBox.Text = combined
end

local function processBuffer()
    if not scanning then return end
    while #buffer >= mode do
        local combined = table.concat(buffer, "", 1, mode)
        buffer = {}
        updatePreview()
        typeAndSubmitCode(combined)
        print("[Galaxy Sniper] Auto-redeemed: " .. combined)
        task.wait(delay)
    end
    updatePreview()
end

local function onAnnouncement(txt)
    if not scanning or not txt or txt == "" then return end
    local msg = txt:match("^%s*(.-)%s*$")
    if msg and #msg > 0 and isValidCode(msg) then
        table.insert(buffer, msg)
        updatePreview()
        if #buffer >= mode then
            task.spawn(processBuffer)
        end
        print("[Galaxy Sniper] Captured: " .. msg .. " (buffer: " .. #buffer .. ")")
    end
end
